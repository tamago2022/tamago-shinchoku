変更前（index.html）：

// フォルダも件数順に出す（フォルダ絞り込み）
const folderList = Object.entries(fc).sort((a,b)=>b[1]-a[1]);
sel.innerHTML = `<option value="">フォルダ：すべて</option>` +
  folderList.map(([f,n])=>`<option value="${f}">${f}（${n}）</option>`).join("");

変更後（index.html）：

// フォルダはEagle側の階層（親 / 子）でグループ化して出す
const groups = {};
Object.keys(fc).sort((a,b)=>a.localeCompare(b,"ja")).forEach(f=>{
  const parts = f.split(" / ");
  const top = parts[0];
  (groups[top] = groups[top] || []).push([f, fc[f], parts.slice(1).join(" / ")]);
});
Object.keys(groups).sort((a,b)=>a.localeCompare(b,"ja")).forEach(top=>{
  const rows = groups[top], total = rows.reduce((s,r)=>s+r[1],0);
  optHtml += `<optgroup label="${top}（${total}）">` +
    rows.map(([f,n,label])=>`<option value="${f}">${label || "（フォルダ全体）"}（${n}）</option>`).join("") +
    `</optgroup>`;
});
