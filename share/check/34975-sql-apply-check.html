-- ごきげん変身屋：申込データの保存テーブル＋写真保存用バケット
--
-- ⚠️ このmigrationはまだ適用しない（supabase db push 禁止）。
-- Kohei確認後に適用する。現時点ではフォームUIのみ実装し、送信は未接続。

CREATE TABLE public.henshin_orders (
  id UUID NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,

  theme TEXT NOT NULL,             -- 選んだ変身テーマのid
  note TEXT,                       -- 補足欄（どんな人・ペットか）
  photo_path TEXT,                 -- storage.objects内のパス（henshin-photosバケット）
  consent BOOLEAN NOT NULL DEFAULT false,

  status TEXT NOT NULL DEFAULT 'pending', -- pending / delivered
  created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now()
);

CREATE INDEX henshin_orders_status_idx
  ON public.henshin_orders(status, created_at DESC);

-- 申込フォームからの新規insertのみ許可。読み出し・更新はservice_role（運営側）のみ。
ALTER TABLE public.henshin_orders ENABLE ROW LEVEL SECURITY;
GRANT INSERT ON public.henshin_orders TO anon, authenticated;
GRANT ALL ON public.henshin_orders TO service_role;

CREATE POLICY "Anyone can submit a henshin order"
  ON public.henshin_orders FOR INSERT
  TO anon, authenticated
  WITH CHECK (true);

-- 写真保存用バケット（非公開：顔写真を含むため一覧公開しない）
INSERT INTO storage.buckets (id, name, public)
VALUES ('henshin-photos', 'henshin-photos', false)
ON CONFLICT (id) DO NOTHING;

CREATE POLICY "Anyone can upload a henshin photo"
  ON storage.objects FOR INSERT
  TO anon, authenticated
  WITH CHECK (bucket_id = 'henshin-photos');
