import { readFileSync } from "node:fs";

const ENDPOINT = "https://vision.googleapis.com/v1/images:annotate";

/** .env.local → .env → process.env の順でキーを探す（値はログに出さない）。 */
export function readVisionApiKey(root) {
  if (process.env.GOOGLE_VISION_API_KEY) return process.env.GOOGLE_VISION_API_KEY;
  if (process.env.GOOGLE_API_KEY) return process.env.GOOGLE_API_KEY;
  for (const file of [".env.local", ".env"]) {
    try {
      const txt = readFileSync(`${root}/${file}`, "utf8");
      for (const name of ["GOOGLE_VISION_API_KEY", "GOOGLE_API_KEY"]) {
        const m = txt.match(new RegExp(`^\\s*${name}\\s*=\\s*(.*)\\s*$`, "m"));
        if (m && m[1]) return m[1].replace(/^["']|["']$/g, "");
      }
    } catch {
      /* 無ければ次を見る */
    }
  }
  return null;
}

/**
 * 画像1枚をVision API（WEB_DETECTION）へ投げ、既存の候補形式
 * `{url, label}[]` に揃えて返す。ブラウザは一切使わない。
 * @returns {Promise<{resultUrl:string|null, aiSummary:string, links:{url:string,label:string}[]}>}
 */
export async function searchByImageVisionApi(apiKey, imagePath) {
  const bytes = readFileSync(imagePath);
  const base64 = bytes.toString("base64");

  const body = {
    requests: [
      {
        image: { content: base64 },
        features: [{ type: "WEB_DETECTION" }],
      },
    ],
  };

  const res = await fetch(`${ENDPOINT}?key=${encodeURIComponent(apiKey)}`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });

  const data = await res.json();
  if (!res.ok) {
    const msg = data?.error?.message || `HTTP ${res.status}`;
    throw new Error(`Vision API エラー: ${msg}`);
  }

  const wd = data?.responses?.[0]?.webDetection;
  if (!wd) {
    return { resultUrl: null, aiSummary: "", links: [] };
  }

  const links = [];
  const seen = new Set();

  // 完全一致・部分一致で画像が載っているページ（元ネタ候補の本体）
  for (const page of wd.pagesWithMatchingImages ?? []) {
    if (!page.url || seen.has(page.url)) continue;
    seen.add(page.url);
    links.push({ url: page.url, label: page.pageTitle || "" });
  }

  // webEntities はURLではなく「これは何か」の説明（Rick Astleyなど）。
  // AI要約の代わりとして文字列にまとめておく。
  const entities = (wd.webEntities ?? [])
    .filter((e) => e.description)
    .slice(0, 8)
    .map((e) => e.description);

  return {
    resultUrl: null, // ブラウザ結果ページという概念が無いため常にnull
    aiSummary: entities.join(" / "),
    links,
  };
}
